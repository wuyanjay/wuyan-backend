## 系统概述

这是一个**极简设计**的图片上传系统，专为博客场景设计。核心理念是：**上传图片 → 返回URL → 完成！**

### 架构图

```mermaid
graph LR
    A[客户端上传] --> B[文件验证]
    B --> C[Sharp图片处理]
    C --> D[保存到uploads/]
    D --> E[写入JSON元数据]
    E --> F[返回URL]
    
    G[查询请求] --> H[读取JSON]
    H --> I[返回结果]
    
    J[删除请求] --> K[删除文件]
    K --> L[更新JSON]
    L --> M[返回成功]
```

### 数据流图

```
上传: 文件 → Sharp压缩 → 保存文件 → 记录元数据 → 返回URL
查询: 请求ID → 读JSON → 返回信息
删除: 请求ID → 删文件 → 删JSON记录 → 完成
列表: 读JSON → 排序分页 → 返回数组
```

### server.js (完整代码)

```javascript
// server.js - 轻量级博客图片上传服务
// 功能: 上传、删除、查询、列表

const Koa = require('koa');
const Router = require('koa-router');
const koaBody = require('koa-body').default;
const serve = require('koa-static');
const sharp = require('sharp');
const fs = require('fs').promises;
const path = require('path');
const crypto = require('crypto');

const app = new Koa();
const router = new Router();

// ==================== 配置 ====================
const CONFIG = {
  uploadDir: './uploads',           // 上传目录
  metaFile: './uploads.json',       // 元数据文件
  maxFileSize: 5 * 1024 * 1024,    // 最大5MB
  maxWidth: 2048,                   // 最大宽度
  maxHeight: 2048,                  // 最大高度
  quality: 85,                      // JPEG质量
};

// ==================== 元数据管理类 ====================
class UploadMeta {
  constructor(filepath) {
    this.filepath = filepath;
    this.data = {};
    this.init();
  }

  // 初始化 - 读取现有数据
  async init() {
    try {
      const content = await fs.readFile(this.filepath, 'utf-8');
      this.data = JSON.parse(content);
    } catch {
      this.data = {};
      await this.save();
    }
  }

  // 保存到文件
  async save() {
    await fs.writeFile(this.filepath, JSON.stringify(this.data, null, 2));
  }

  // 添加新记录
  async add(id, meta) {
    this.data[id] = {
      ...meta,
      uploadedAt: new Date().toISOString(),
    };
    await this.save();
    return this.data[id];
  }

  // 获取单条记录
  get(id) {
    return this.data[id];
  }

  // 删除记录
  async delete(id) {
    delete this.data[id];
    await this.save();
  }

  // 列表查询 (分页)
  list(limit = 100, offset = 0) {
    const items = Object.entries(this.data)
      .map(([id, meta]) => ({ id, ...meta }))
      .sort((a, b) => new Date(b.uploadedAt) - new Date(a.uploadedAt));
    
    return {
      items: items.slice(offset, offset + limit),
      total: items.length,
      limit,
      offset,
    };
  }
}

// 初始化元数据管理器
const meta = new UploadMeta(CONFIG.metaFile);

// 确保上传目录存在
fs.mkdir(CONFIG.uploadDir, { recursive: true });

// ==================== 图片处理函数 ====================
async function processImage(buffer) {
  return sharp(buffer)
    .resize({
      width: CONFIG.maxWidth,
      height: CONFIG.maxHeight,
      fit: 'inside',                // 保持宽高比
      withoutEnlargement: true,     // 不放大小图
    })
    .jpeg({
      quality: CONFIG.quality,
      progressive: true,            // 渐进式加载
    })
    .toBuffer();
}

// ==================== 路由定义 ====================

// 1. 上传图片
router.post('/upload', async (ctx) => {
  const file = ctx.request.files?.file;
  
  // 验证文件存在
  if (!file) {
    ctx.status = 400;
    ctx.body = { success: false, error: 'No file uploaded' };
    return;
  }

  // 验证文件类型
  const allowedTypes = ['image/jpeg', 'image/jpg', 'image/png', 'image/webp'];
  if (!allowedTypes.includes(file.mimetype || file.type)) {
    ctx.status = 400;
    ctx.body = { success: false, error: 'Invalid file type. Allowed: JPEG, PNG, WebP' };
    return;
  }

  try {
    // 读取原始文件
    const buffer = await fs.readFile(file.filepath);
    
    // 处理图片 (压缩优化)
    const processed = await processImage(buffer);
    
    // 生成唯一ID和文件名
    const id = crypto.randomUUID();
    const ext = path.extname(file.originalFilename || file.name) || '.jpg';
    const filename = `${id}${ext}`;
    const filepath = path.join(CONFIG.uploadDir, filename);
    
    // 保存处理后的文件
    await fs.writeFile(filepath, processed);
    
    // 保存元数据
    const imageInfo = await meta.add(id, {
      url: `/uploads/${filename}`,
      filename: file.originalFilename || file.name,
      size: processed.length,
      originalSize: buffer.length,
      mimeType: file.mimetype || file.type,
    });
    
    // 返回成功响应
    ctx.body = {
      success: true,
      id,
      url: imageInfo.url,
      size: imageInfo.size,
    };
  } catch (error) {
    console.error('Upload error:', error);
    ctx.status = 500;
    ctx.body = { success: false, error: error.message };
  } finally {
    // 清理临时文件
    await fs.unlink(file.filepath).catch(() => {});
  }
});

// 2. 删除图片
router.delete('/upload/:id', async (ctx) => {
  const { id } = ctx.params;
  const info = meta.get(id);
  
  // 验证图片存在
  if (!info) {
    ctx.status = 404;
    ctx.body = { success: false, error: 'Image not found' };
    return;
  }

  try {
    // 删除文件
    const filename = path.basename(info.url);
    const filepath = path.join(CONFIG.uploadDir, filename);
    await fs.unlink(filepath);
    
    // 删除元数据
    await meta.delete(id);
    
    ctx.body = { success: true };
  } catch (error) {
    console.error('Delete error:', error);
    ctx.status = 500;
    ctx.body = { success: false, error: error.message };
  }
});

// 3. 获取图片信息
router.get('/upload/:id', async (ctx) => {
  const { id } = ctx.params;
  const info = meta.get(id);
  
  if (!info) {
    ctx.status = 404;
    ctx.body = { success: false, error: 'Image not found' };
    return;
  }

  ctx.body = {
    success: true,
    data: { id, ...info },
  };
});

// 4. 列表查询
router.get('/uploads', async (ctx) => {
  const limit = parseInt(ctx.query.limit) || 100;
  const offset = parseInt(ctx.query.offset) || 0;
  
  const result = meta.list(limit, offset);
  
  ctx.body = {
    success: true,
    ...result,
  };
});

// 5. 健康检查
router.get('/health', async (ctx) => {
  ctx.body = { 
    status: 'ok', 
    timestamp: new Date().toISOString(),
    stats: {
      totalImages: Object.keys(meta.data).length,
    }
  };
});

// ==================== 中间件配置 ====================

// 错误处理
app.use(async (ctx, next) => {
  try {
    await next();
  } catch (err) {
    console.error('Error:', err);
    ctx.status = err.status || 500;
    ctx.body = {
      success: false,
      error: err.message,
    };
  }
});

// 请求日志
app.use(async (ctx, next) => {
  const start = Date.now();
  await next();
  const ms = Date.now() - start;
  console.log(`${ctx.method} ${ctx.url} - ${ctx.status} - ${ms}ms`);
});

// 请求体解析
app.use(koaBody({
  multipart: true,
  formidable: {
    maxFileSize: CONFIG.maxFileSize,
    keepExtensions: true,
  },
}));

// 静态文件服务
app.use(serve(CONFIG.uploadDir));

// 应用路由
app.use(router.routes());
app.use(router.allowedMethods());

// ==================== 启动服务 ====================
const PORT = process.env.PORT || 3000;

app.listen(PORT, () => {
  console.log('');
  console.log('🚀 Lightweight Image Upload Server');
  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  console.log(`📍 Server:     http://localhost:${PORT}`);
  console.log(`📁 Upload dir: ${CONFIG.uploadDir}`);
  console.log(`📊 Max size:   ${CONFIG.maxFileSize / 1024 / 1024}MB`);
  console.log(`🖼️  Max dims:   ${CONFIG.maxWidth}x${CONFIG.maxHeight}px`);
  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━');
  console.log('');
  console.log('📖 API Endpoints:');
  console.log(`   POST   /upload           Upload image`);
  console.log(`   DELETE /upload/:id       Delete image`);
  console.log(`   GET    /upload/:id       Get image info`);
  console.log(`   GET    /uploads          List images`);
  console.log(`   GET    /uploads/:file    View image`);
  console.log(`   GET    /health           Health check`);
  console.log('');
});

module.exports = app;
```

### 方式四: Nginx反向代理

**nginx.conf**
```nginx
server {
    listen 80;
    server_name images.yourdomain.com;

    # 上传接口
    location /upload {
        proxy_pass http://localhost:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        client_max_body_size 10M;
    }

    # 静态文件直接服务 (性能更好)
    location /uploads/ {
        alias /path/to/uploads/;
        expires 1y;
        add_header Cache-Control "public, immutable";
        
        # 防盗链 (可选)
        valid_referers none blocked yourdomain.com *.yourdomain.com;
        if ($invalid_referer) {
            return 403;
        }
    }
}
```

## 前端集成示例

### React组件

```jsx
import React, { useState } from 'react';

function ImageUpload() {
  const [uploading, setUploading] = useState(false);
  const [imageUrl, setImageUrl] = useState('');
  const [error, setError] = useState('');

  const handleUpload = async (e) => {
    const file = e.target.files[0];
    if (!file) return;

    // 验证文件类型
    if (!file.type.startsWith('image/')) {
      setError('请选择图片文件');
      return;
    }

    // 验证文件大小 (5MB)
    if (file.size > 5 * 1024 * 1024) {
      setError('文件大小不能超过5MB');
      return;
    }

    setUploading(true);
    setError('');

    const formData = new FormData();
    formData.append('file', file);

    try {
      const response = await fetch('http://localhost:3000/upload', {
        method: 'POST',
        body: formData,
      });

      const data = await response.json();

      if (data.success) {
        setImageUrl(`http://localhost:3000${data.url}`);
        console.log('上传成功，图片ID:', data.id);
      } else {
        setError(data.error);
      }
    } catch (error) {
      setError('上传失败: ' + error.message);
    } finally {
      setUploading(false);
    }
  };

  return (
    <div>
      <input
        type="file"
        onChange={handleUpload}
        accept="image/*"
        disabled={uploading}
      />
      
      {uploading && <p>上传中...</p>}
      {error && <p style={{ color: 'red' }}>{error}</p>}
      {imageUrl && (
        <div>
          <p>上传成功！</p>
          <img src={imageUrl} alt="Uploaded" style={{ maxWidth: '100%' }} />
          <p>图片URL: {imageUrl}</p>
        </div>
      )}
    </div>
  );
}

export default ImageUpload;
```

    <p v-if="error" class="error">{{ error }}</p>

    <div v-if="imageUrl">
      <p>上传成功！</p>
      <img :src="imageUrl" alt="Uploaded" style="max-width: 100%;" />
      <p>图片URL: {{ imageUrl }}</p>
    </div>
  </div>
</template>

<script>
export default {
  data() {
    return {
      uploading: false,
      imageUrl: '',
      error: '',
    };
  },
  methods: {
    async handleUpload(e) {
      const file = e.target.files[0];
      if (!file) return;

      // 验证文件类型
      if (!file.type.startsWith('image/')) {
        this.error = '请选择图片文件';
        return;
      }

      // 验证文件大小
      if (file.size > 5 * 1024 * 1024) {
        this.error = '文件大小不能超过5MB';
        return;
      }

      this.uploading = true;
      this.error = '';

      const formData = new FormData();
      formData.append('file', file);

      try {
        const response = await fetch('http://localhost:3000/upload', {
          method: 'POST',
          body: formData,
        });

        const data = await response.json();

        if (data.success) {
          this.imageUrl = `http://localhost:3000${data.url}`;
          console.log('上传成功，图片ID:', data.id);
        } else {
          this.error = data.error;
        }
      } catch (error) {
        this.error = '上传失败: ' + error.message;
      } finally {
        this.uploading = false;
      }
    },
  },
};
</script>

<style scoped>
.error {
  color: red;
}
</style>

### Q5: 如何限制上传频率？

**A**: 使用内存限流：

```javascript
// 简单的速率限制器
const uploadLimiter = new Map();

router.post('/upload', async (ctx) => {
  const ip = ctx.ip;
  const now = Date.now();
  const limit = uploadLimiter.get(ip) || [];
  
  // 清理1分钟前的记录
  const recent = limit.filter(time => now - time < 60000);
  
  // 限制每分钟10次
  if (recent.length >= 10) {
    ctx.status = 429;
    ctx.body = { success: false, error: 'Too many uploads. Please wait.' };
    return;
  }
  
  uploadLimiter.set(ip, [...recent, now]);
  
  // ... 上传逻辑
});
```

### Q7: 如何处理重复上传？

**A**: 使用文件哈希去重：

const crypto = require('crypto');

// 计算文件哈希
function getFileHash(buffer) {
  return crypto.createHash('md5').update(buffer).digest('hex');
}

router.post('/upload', async (ctx) => {
  const buffer = await fs.readFile(file.filepath);

      success: true,
      id: existing.id,
      url: existing.url,
      duplicate: true,

    return;

});

### Q8: 如何定期清理无用图片？

**A**: 创建清理脚本：

```javascript
// cleanup.js
const fs = require('fs').promises;
const path = require('path');

async function cleanup() {
  const meta = JSON.parse(await fs.readFile('./uploads.json', 'utf-8'));
  const uploadDir = './uploads';
  
  // 获取所有文件
  const files = await fs.readdir(uploadDir);
  
  // 获取所有记录中的文件名
  const validFiles = Object.values(meta).map(item => path.basename(item.url));
  
  // 删除无记录的文件
  for (const file of files) {
    if (!validFiles.includes(file)) {
      await fs.unlink(path.join(uploadDir, file));
      console.log('Deleted orphan file:', file);
    }
  }
}

cleanup();
```

```bash

```
